sudo chmod 755 /run/mariadb

sudo pkill -f mariadbd
sudo pkill -f mysqld

# 6. Remove old socket & PID files (optional but safe)
sudo rm -f /var/lib/mysql/mysql.sock
sudo rm -f /var/lib/mysql/mariadb.pid

# 7. Start MariaDB manually
sudo mysqld_safe --user=mysql &

# 8. Verify MariaDB is running
